cmake_minimum_required(VERSION 3.20)
project(TMT VERSION 0.1.0 LANGUAGES C)
if(NOT UNIX)
  message(FATAL_ERROR "TMT preparation currently requires POSIX process and library loading APIs")
endif()
include(GNUInstallDirs)
include(CTest)
option(TMT_BUILD_FORTRAN "Build the Fortran interface" ON)
option(TMT_SANITIZE "Address and undefined behavior sanitizers" OFF)
set(TMT_DEPENDENCY_DIR "${CMAKE_BINARY_DIR}/_deps" CACHE PATH "Extracted source snapshots")
set(TMT_REPOSITORIES "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Local source repositories")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/dependencies.py"
  --output "${TMT_DEPENDENCY_DIR}" --repositories "${TMT_REPOSITORIES}"
  RESULT_VARIABLE dependency_result)
if(NOT dependency_result EQUAL 0)
  message(FATAL_ERROR "Dependency extraction failed")
endif()
set(TURBOTUV_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(TURBOTUV_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(saved_testing ${BUILD_TESTING})
set(BUILD_TESTING OFF)
add_subdirectory("${TMT_DEPENDENCY_DIR}/turbotuv" "${CMAKE_BINARY_DIR}/turbotuv")
set(BUILD_TESTING ${saved_testing})
find_package(OpenMP REQUIRED COMPONENTS C)
file(GLOB fable_sources CONFIGURE_DEPENDS "${TMT_DEPENDENCY_DIR}/fable/src/*.c")
add_library(tmt_fable SHARED ${fable_sources})
target_include_directories(tmt_fable PUBLIC
  $<BUILD_INTERFACE:${TMT_DEPENDENCY_DIR}/fable/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/tmt/fable>)
target_compile_definitions(tmt_fable PRIVATE TMC_WIDTH=2 TMC_DEFAULT_GROUP_SIZE=32)
target_link_libraries(tmt_fable PUBLIC OpenMP::OpenMP_C m)
set_target_properties(tmt_fable PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
file(GLOB_RECURSE fable_headers CONFIGURE_DEPENDS "${TMT_DEPENDENCY_DIR}/fable/include/*.h" "${TMT_DEPENDENCY_DIR}/fable/src/*.h")
set(fable_identity "")
foreach(source IN LISTS fable_sources fable_headers)
  file(SHA256 "${source}" source_hash)
  string(APPEND fable_identity ";${source_hash}")
endforeach()
string(SHA256 TMT_BUILD_ID "${fable_identity};${CMAKE_C_COMPILER_ID};${CMAKE_C_COMPILER_VERSION};${CMAKE_SYSTEM_NAME};${CMAKE_SYSTEM_PROCESSOR};${CMAKE_SIZEOF_VOID_P};${CMAKE_BUILD_TYPE};${CMAKE_C_FLAGS};${CMAKE_C_FLAGS_RELEASE};${CMAKE_C_FLAGS_DEBUG};${CMAKE_C_FLAGS_RELWITHDEBINFO};${CMAKE_C_FLAGS_MINSIZEREL};${CMAKE_OSX_SYSROOT};${CMAKE_OSX_ARCHITECTURES};${OpenMP_C_FLAGS};${TMT_SANITIZE};WIDTH2;GROUP32;ABI1")
configure_file(cmake/paths.h.in "${CMAKE_BINARY_DIR}/generated/paths.h" @ONLY)
add_library(tmt SHARED src/common/common.c src/tuv/objects.c src/tuv/model.c src/prepare/prepare.c src/micm/micm.c)
add_library(musica::musica ALIAS tmt)
target_include_directories(tmt PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  PRIVATE src/common "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(tmt PRIVATE _POSIX_C_SOURCE=200809L)
target_link_libraries(tmt PRIVATE turbotuv tmt_fable ${CMAKE_DL_LIBS})
set_target_properties(tmt PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(APPLE)
  set_target_properties(tmt tmt_fable PROPERTIES INSTALL_RPATH "@loader_path" BUILD_WITH_INSTALL_RPATH TRUE)
else()
  set_target_properties(tmt tmt_fable PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(tmt PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(TMT_SANITIZE)
  foreach(target tmt tmt_fable turbotuv turbotuv_yaml)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
install(TARGETS tmt tmt_fable EXPORT TMTTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY "${TMT_DEPENDENCY_DIR}/fable/include/" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/tmt/fable)
install(DIRECTORY "${TMT_DEPENDENCY_DIR}/fable/include/" DESTINATION ${CMAKE_INSTALL_DATADIR}/tmt/fable/include)
install(DIRECTORY "${TMT_DEPENDENCY_DIR}/fable/tools/" DESTINATION ${CMAKE_INSTALL_DATADIR}/tmt/fable/tools
  FILES_MATCHING PATTERN "*.py")
install(FILES tools/normalize.py tools/stage_mechanism.py DESTINATION ${CMAKE_INSTALL_DATADIR}/tmt/tools)
install(DIRECTORY "${TMT_DEPENDENCY_DIR}/musica/configs/" DESTINATION ${CMAKE_INSTALL_DATADIR}/tmt/configs)
set(TMT_PYTHON_INSTALL_DIR "${CMAKE_INSTALL_LIBDIR}/python${Python3_VERSION_MAJOR}.${Python3_VERSION_MINOR}/site-packages" CACHE STRING "Python package destination relative to install prefix")
install(DIRECTORY python/musica python/tmt DESTINATION ${TMT_PYTHON_INSTALL_DIR}
  PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES LICENSE NOTICE dependencies.lock DESTINATION ${CMAKE_INSTALL_DATADIR}/tmt)
install(FILES "${TMT_DEPENDENCY_DIR}/musica/LICENSE" DESTINATION ${CMAKE_INSTALL_DATADIR}/tmt/licenses RENAME MUSICA.txt)
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/tmt/examples)
install(FILES tests/configs/decay.json DESTINATION ${CMAKE_INSTALL_DATADIR}/tmt/examples)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/TMTConfig.cmake.in "${CMAKE_BINARY_DIR}/TMTConfig.cmake"
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TMT)
write_basic_package_version_file("${CMAKE_BINARY_DIR}/TMTConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT TMTTargets NAMESPACE TMT:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TMT)
install(FILES "${CMAKE_BINARY_DIR}/TMTConfig.cmake" "${CMAKE_BINARY_DIR}/TMTConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TMT)
configure_file(cmake/tmt.pc.in "${CMAKE_BINARY_DIR}/tmt.pc" @ONLY)
configure_file(cmake/tmt-fortran.pc.in "${CMAKE_BINARY_DIR}/tmt-fortran.pc" @ONLY)
install(FILES "${CMAKE_BINARY_DIR}/tmt.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)

if(TMT_BUILD_FORTRAN)
  enable_language(Fortran)
  file(GLOB fortran_tuv CONFIGURE_DEPENDS fortran/tuvx/*.F90)
  add_library(tmt_fortran fortran/util.F90 fortran/musica.F90 fortran/micm/state.F90 fortran/micm/micm.F90 ${fortran_tuv})
  add_library(musica::musica-fortran ALIAS tmt_fortran)
  set_target_properties(tmt_fortran PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
  target_include_directories(tmt_fortran PUBLIC
    $<BUILD_INTERFACE:${CMAKE_BINARY_DIR}/modules>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/tmt/modules>)
  target_link_libraries(tmt_fortran PUBLIC tmt)
  if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
    target_compile_options(tmt_fortran PRIVATE -ffree-line-length-none)
  endif()
  install(TARGETS tmt_fortran EXPORT TMTTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(DIRECTORY "${CMAKE_BINARY_DIR}/modules/" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/tmt/modules)
  install(FILES "${CMAKE_BINARY_DIR}/tmt-fortran.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
endif()

  add_test(NAME tuv_c COMMAND test_tuv "${TMT_DEPENDENCY_DIR}/turbotuv/examples/tuv_5_4.json"

    "${TMT_DEPENDENCY_DIR}/turbotuv/examples/tuv_5_4.yml"

  if(TMT_BUILD_FORTRAN)
    file(READ "${TMT_DEPENDENCY_DIR}/musica/fortran/test/unit/util.F90" util_fortran_test)

    set_tests_properties(common_fortran_upstream PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/reference-run")
    file(READ "${TMT_DEPENDENCY_DIR}/musica/fortran/test/integration/test_micm_api.F90" micm_fortran_test)
    string(REPLACE "micm/util/error.hpp" "musica/micm_errors.h" micm_fortran_test "${micm_fortran_test}")
    string(REPLACE "musica/utils/error.hpp" "musica/utils/error.h" micm_fortran_test "${micm_fortran_test}")
    file(WRITE "${CMAKE_BINARY_DIR}/generated/test_micm_api.F90" "${micm_fortran_test}")
    add_executable(test_micm_fortran "${CMAKE_BINARY_DIR}/generated/test_micm_api.F90")

      ENVIRONMENT "TMT_CACHE=${CMAKE_BINARY_DIR}/test mechanism cache")

  add_test(NAME tuv_python COMMAND "${Python3_EXECUTABLE}" -m pytest "${CMAKE_CURRENT_SOURCE_DIR}/tests/python/test_tuv.py" -q)

  set(tuv_unit_dir "${TMT_DEPENDENCY_DIR}/musica/python/test/unit/tuvx")
  add_test(NAME tuv_python_upstream COMMAND "${Python3_EXECUTABLE}" -m pytest "--confcutdir=${tuv_unit_dir}"
    "${tuv_unit_dir}/test_grid.py" "${tuv_unit_dir}/test_grid_map.py"
    "${tuv_unit_dir}/test_profile.py" "${tuv_unit_dir}/test_profile_map.py"

  set(micm_unit_dir "${TMT_DEPENDENCY_DIR}/musica/python/test/unit/micm")
